+def independent_standard_params(x):
+    """Two-pass corrected population variance and near-constant scale.
+
+    Reimplement the pinned StandardScaler arithmetic without calling its
+    fitting/variance helpers. Count non-NaN samples, retain float64, then use
+    the theoretical variance-error bound documented by sklearn 1.6.0.
+    """
+    x=np.asarray(x,dtype=np.float64)
+    count=np.count_nonzero(~np.isnan(x))
+    if count==0:
+        return np.nan,np.nan,np.nan
+    center=np.nansum(x)/count
+    residual=x-center
+    correction=np.nansum(residual)
+    variance=(np.nansum(residual*residual)-correction*correction/count)/count
+    eps=np.finfo(np.float64).eps
+    constant=variance <= count*eps*variance+(count*center*eps)**2
+    scale=1. if constant else np.sqrt(variance)
+    return center,variance,scale
